\section{The Cloud}


\begin{frame}{What is the cloud?}{Official definition by ISO}
  A paradigm for enabling network access to a scalable and elastic pool of shareable physical or virtual resources with self-service provisioning and administration on demand.

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.9\textwidth,height=0.6\textheight,keepaspectratio]{module-b4-cloud-computing}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{What is the cloud?}{My simplified definition}
  A computer or a specific computing service that we can access remotely over the internet.

  \begin{columns}[T]
  \begin{column}{0.58\textwidth}
    \begin{itemize}
      \item Usually, the hardware behind the cloud sits in \textbf{data centers}
      \item Technically, a cloud computer can be used for anything a personal computer can do, except for things that strictly require physical access to the hardware
      \item The hardware and part of the software are owned and managed by someone else; we are renting them, subject to specific terms
    \end{itemize}
  \end{column}

  \begin{column}{0.4\textwidth}
    \begin{figure}[htpb]
        \begin{center}
            \includegraphics[width=\linewidth]{cern-data-centre}
            \caption*{Inside the CERN Data Centre, home to CERN's private cloud}
        \end{center}
    \end{figure}
  \end{column}
  \end{columns}
\note{
}
\end{frame}


\begin{frame}{Why call it ``the cloud''?}
  \begin{itemize}
    \item Source: network diagrams, where a \textit{fluffy cloud symbol} is used to convey ``somewhere out there is the network, the details do not matter''
    \item The symbolic meaning of the cloud gradually shifted to represent ``computing resources at the other end of the internet''
  \end{itemize}

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.9\textwidth,height=0.56\textheight,keepaspectratio]{module-b4-network-diagram}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{How the cloud came into existence}
  IT companies rent out computing resources they accumulated to handle peak hours that are idle most of the time.

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.9\textwidth,height=0.56\textheight,keepaspectratio]{amazon-november-traffic}
          \caption*{Capacity bought for the November peak (red line) leaves 76\% of it idle. Source: AWS, \textit{Migrating Amazon.com to AWS} (2011)}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{The cloud as a business model}
  \begin{block}{Benefits for us consumers}
    \begin{itemize}
      \item A new computer can be set up and ready to use within minutes
      \item Pay for what we use, mitigate the need for purchasing hardware up front
      \item We do not need to worry about the management of the hardware, network infrastructure, electricity, etc.
    \end{itemize}
  \end{block}

  \begin{block}{Downsides for us consumers}
    We never own the hardware and are subject to the cloud provider's pricing, terms, and conditions
  \end{block}
\note{
}
\end{frame}


\begin{frame}{Virtualization: The technology powering the cloud}
  \begin{itemize}
    \item Virtualization can slice a powerful computer's hardware (e.g., CPU cores, memory, hard drive) or computing resources in general (e.g., CPU core hours) into small pieces (virtual machines)
    \item Each virtual machine can have its own operating system and everything above, isolated from other virtual machines
  \end{itemize}
  This allows us to rent cloud computers that fit exactly into our needs.

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.9\textwidth,height=0.46\textheight,keepaspectratio]{module-b4-vm-vs-container-vm}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{Virtualization vs. containerization}
  \begin{itemize}
    \item \textbf{Virtualization:} virtualizes the entire hardware, allowing each virtual machine to run its own operating system
    \item \textbf{Containerization:} all containers share the host computer's operating system's kernel
  \end{itemize}

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.9\textwidth,height=0.55\textheight,keepaspectratio]{module-b2-vm-vs-container}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{Common cloud providers}
    \begin{block}{Hyperscalers}
        \begin{itemize}
            \item AWS: the original, biggest catalog
            \item GCP: focus on data and ML services, competitive GPU pricing
            \item Azure: natural fit for the Microsoft ecosystem
        \end{itemize}
    \end{block}

    \begin{block}{Smaller providers}
        \begin{itemize}
            \item DigitalOcean, Linode, Vultr: clean UIs, predictable pricing
            \item Lambda, CoreWeave, RunPod: GPU rentals at lower prices
            \item Hetzner, OVHcloud, Scaleway: EU-based, often cheap basic VMs
        \end{itemize}
    \end{block}

\end{frame}

\section{Public Endpoints}


\begin{frame}{Current status of our API endpoints}
  They are publicly reachable by clients over the internet, but not in a shape we want.

  \begin{itemize}
    \item \textbf{Raw IP address:} The endpoint is only reachable through our cloud computer's IP address
      \begin{itemize}
        \item An IP address is hard to remember and looks unprofessional
        \item An IP address cannot transfer to a new virtual machine if we ever need to migrate our service
      \end{itemize}
    \item \textbf{Unencrypted HTTP traffic:} The full content of the requests clients send to our API server, and the responses it sends back, is visible to anyone on the same network path
      \begin{itemize}
        \item A huge security loophole since the requests and responses might contain highly personal or confidential information
        \item Most modern clients and web browsers block unencrypted HTTP traffic altogether
      \end{itemize}
  \end{itemize}

  \vspace{1em}
  \live{The reading material (blog site) deployed on a cloud virtual machine.}
\note{
}
\end{frame}

\begin{frame}{Give a domain to our API endpoints}{Register a domain}
  \begin{itemize}
    \item Unique domains can be purchased from registrars like Cloudflare and Amazon Route 53
    \item Free domains can be obtained from places like DuckDNS for learning purposes
  \end{itemize}

\begin{frame}{Enable HTTPS for our API endpoints}{What is HTTPS?}
  HTTPS (HTTP over TLS) encrypts HTTP traffic, and only the clients and our server can decrypt the data.

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.9\textwidth,height=0.66\textheight,keepaspectratio]{module-b4-https}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{Enable HTTPS for our API endpoints}{Obtain certificates for our domain}
  For a domain we own, we can request an HTTPS certificate from free services like \textit{Let's Encrypt}.

  \begin{enumerate}
    \item Point the domain to our virtual machine with a DNS record
    \item Install an ACME client, such as \textit{Certbot}, on the virtual machine
    \item Request a certificate for the domain; the ACME client proves to Let's Encrypt that we control the domain
    \item Receive the certificate and its private key, stored on the virtual machine
  \end{enumerate}
\note{
}
\end{frame}


\begin{frame}{Enable HTTPS for our API endpoints}{Wrap our API endpoints behind a reverse proxy}
  A reverse proxy (e.g., Nginx) faces the internet. It holds the certificate and private key, accepts HTTPS connections from clients, decrypts them, and forwards the requests as plain HTTP to our API endpoints.

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.9\textwidth,height=0.55\textheight,keepaspectratio]{module-b4-reverse-proxy}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{Enable HTTPS for our API endpoints}{Simpler solutions}
  \begin{block}{Reverse proxies with built-in certificate management}
    \begin{itemize}
      \item \textbf{Caddy:} obtains and renews certificates automatically for every domain it serves
      \item \textbf{Traefik:} discovers our containers via Docker labels and gets certificates for them
      \item \textbf{Nginx Proxy Manager:} a web UI on top of Nginx to set up proxies and certificates
    \end{itemize}
  \end{block}

  \begin{block}{One-click tunnels}
    \begin{itemize}
      \item \textbf{Cloudflare Tunnel:} a connector on our virtual machine opens an outbound tunnel to Cloudflare, which serves our endpoints over HTTPS
      \item Similar services: ngrok, Tailscale Funnel
    \end{itemize}
  \end{block}
\note{
}
\end{frame}
